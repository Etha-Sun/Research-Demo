# Re-implementation (copied logic) of topoverus/matcher_v2.py + graph.py dependency_score, for a worked example.
import re
TOKEN = re.compile(r"[A-Za-z_][A-Za-z_0-9]*|\d+(?:[A-Za-z_][A-Za-z_0-9]*)?|==|!=|<=|>=|&&|\|\||::|->|=>|[^\s]")
def normalized(t):
    t = re.sub(r"//[^\n]*", "", t); return " ".join(t.strip().rstrip(",").split())
def sim(a,b):
    A,B=set(TOKEN.findall(normalized(a))),set(TOKEN.findall(normalized(b)))
    return len(A&B)/len(A|B)
def match(cands, canon, th=0.6):
    assigned, used = {}, set()
    for p,c in enumerate(cands):
        ex=[i for i,v in enumerate(canon) if i not in used and normalized(c)==normalized(v)]
        if ex: assigned[p]={"position":p,"canonical":ex[0],"kind":"exact","similarity":1.0}; used.add(ex[0])
    for p,c in enumerate(cands):
        if p in assigned: continue
        av=[(sim(c,v),i) for i,v in enumerate(canon) if i not in used]
        if av:
            s,i=max(av,key=lambda x:(x[0],-x[1]))
            if s>=th: assigned[p]={"position":p,"canonical":i,"kind":"fuzzy","similarity":round(s,3)}; used.add(i)
    return [assigned[p] for p in sorted(assigned)]
def comps(n,edges):
    r=[[i==j for j in range(n)] for i in range(n)]
    for a,b in edges: r[a][b]=True
    for k in range(n):
        for i in range(n):
            for j in range(n): r[i][j]|=r[i][k] and r[k][j]
    return [min(j for j in range(n) if r[i][j] and r[j][i]) for i in range(n)]
def score(cands, canon, edges):
    m=match(cands,canon); cov=len(m)/len(canon); c=comps(len(canon),edges)
    cond={(c[a],c[b]) for a,b in edges if c[a]!=c[b]}
    pos={}
    for it in m: pos.setdefault(c[it["canonical"]], it["position"])
    cons=[(a,b) for a,b in cond if a in pos and b in pos]
    sat=(sum(pos[a]<pos[b] for a,b in cons)/len(cons)) if cons else 1.0
    return dict(matches=[(x["position"],"I%d"%(x["canonical"]+1),x["kind"],x["similarity"]) for x in m],
                alpha=round(cov,3), considered=len(cons), satisfied=sum(pos[a]<pos[b] for a,b in cons), beta=round(sat,3), R_dep=round(cov*sat,3))
